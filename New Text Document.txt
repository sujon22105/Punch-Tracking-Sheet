<script>
    const form = document.getElementById('punchForm');
    const tableBody = document.getElementById('tableBody');

    form.addEventListener('submit', function(e) {
        e.preventDefault();

        const punchId = document.getElementById('punchId').value;
        const receivingDate = document.getElementById('receivingDate').value;
        const upperCountSet = document.getElementById('upperCountSet').value;
        const lowerCountSet = document.getElementById('lowerCountSet').value;
        const upperProduced = parseFloat(document.getElementById('upperProduced').value);
        const lowerProduced = parseFloat(document.getElementById('lowerProduced').value);
        const lastInspectionDate = document.getElementById('lastInspectionDate').value;

        // Calculations (1.5 Million limit or 1 Year check)
        const totalProduced = upperProduced + lowerProduced;
        
        // Date calculation for 1 year (365 days)
        const lastInsp = new Date(lastInspectionDate);
        const today = new Date();
        const diffTime = Math.abs(today - lastInsp);
        const diffDays = Math.ceil(diffTime / (1000 * 60 * 60 * 24));

        let notification = "Safe";
        let statusClass = "alert-green";

        if (totalProduced >= 1.5 || diffDays >= 335) { // 335 days means 30 days before 1 year (365 days)
            notification = "Inspection Due / Limit Reached!";
            statusClass = "alert-red";
        }

        const row = document.createElement('tr');
        row.innerHTML = `
            <td>${punchId}</td>
            <td>${receivingDate}</td>
            <td>U: ${upperCountSet} <br> L: ${lowerCountSet}</td>
            <td>${totalProduced.toFixed(2)}M</td>
            <td>U: ${upperProduced}M <br> L: ${lowerProduced}M</td>
            <td>${lastInspectionDate}</td>
            <td>Days Passed: ${diffDays} days</td>
            <td class="${statusClass}">${notification}</td>
        `;

        tableBody.appendChild(row);
        form.reset();
    });
</script>
